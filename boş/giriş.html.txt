<!-- Firebase SDK -->
<script type="module">
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";
import {
  getAuth, signInWithEmailAndPassword, signOut, onAuthStateChanged
} from "https://www.gstatic.com/firebasejs/10.12.0/firebase-auth.js";
import {
  getFirestore, collection, addDoc, query, orderBy, onSnapshot, serverTimestamp
} from "https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js";
import {
  getStorage, ref, uploadBytesResumable, getDownloadURL
} from "https://www.gstatic.com/firebasejs/10.12.0/firebase-storage.js";

// ⚠️ CRM / Fon Takip'te kullandığın firebaseConfig ile AYNI değerleri buraya koy
const firebaseConfig = {
  apiKey: "...",
  authDomain: "...",
  projectId: "...",
  storageBucket: "...",
  messagingSenderId: "...",
  appId: "..."
};

const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const storage = getStorage(app);

const CATS = ["Gündem","Ekonomi","Spor","Doğa","İnsan","Sanat","Teknoloji","Sembol"];
let activeCat = "Tümü";
let allVideos = [];

window.login = async function () {
  const email = document.getElementById("email").value;
  const password = document.getElementById("password").value;
  document.getElementById("authError").textContent = "";
  try {
    await signInWithEmailAndPassword(auth, email, password);
  } catch (e) {
    document.getElementById("authError").textContent = "Giriş başarısız: " + e.message;
  }
};

window.logout = function () { signOut(auth); };

onAuthStateChanged(auth, (user) => {
  document.getElementById("authGate").style.display = user ? "none" : "flex";
  document.getElementById("appShell").style.display = user ? "block" : "none";
  if (user) initTabs(), listenVideos();
});

function initTabs() {
  const tabsEl = document.getElementById("tabs");
  tabsEl.innerHTML = "";
  ["Tümü", ...CATS].forEach(cat => {
    const btn = document.createElement("button");
    btn.className = "tab" + (cat === activeCat ? " active" : "");
    btn.textContent = cat;
    btn.onclick = () => { activeCat = cat; initTabs(); renderGrid(); };
    tabsEl.appendChild(btn);
  });
}

function listenVideos() {
  const q = query(collection(db, "videos"), orderBy("createdAt", "desc"));
  onSnapshot(q, (snap) => {
    allVideos = snap.docs.map(d => ({ id: d.id, ...d.data() }));
    renderGrid();
  });
}

function renderGrid() {
  const grid = document.getElementById("grid");
  const filtered = activeCat === "Tümü" ? allVideos : allVideos.filter(v => v.category === activeCat);
  if (!filtered.length) {
    grid.innerHTML = '<div class="empty">Bu kategoride henüz video yok</div>';
    return;
  }
  grid.innerHTML = "";
  filtered.forEach(v => {
    const card = document.createElement("div");
    card.className = "card";
    card.innerHTML = `
      <video src="${v.url}" controls preload="metadata"></video>
      <div class="card-body">
        <div class="card-title">${v.title || "(başlıksız)"}</div>
        <div class="card-cat">${v.category}</div>
        ${v.notes ? `<div class="card-notes">${v.notes}</div>` : ""}
      </div>`;
    grid.appendChild(card);
  });
}

window.uploadVideo = async function () {
  const fileInput = document.getElementById("file");
  const file = fileInput.files[0];
  const title = document.getElementById("title").value.trim();
  const category = document.getElementById("category").value;
  const notes = document.getElementById("notes").value.trim();
  const statusEl = document.getElementById("uploadStatus");
  const progressWrap = document.getElementById("progressWrap");
  const progressBar = document.getElementById("progressBar");

  if (!file) { statusEl.textContent = "Önce bir video dosyası seç."; return; }

  const safeName = Date.now() + "-" + file.name.replace(/[^a-zA-Z0-9.\-_]/g, "_");
  const storagePath = `reels/${category}/${safeName}`;
  const storageRef = ref(storage, storagePath);
  const task = uploadBytesResumable(storageRef, file);

  progressWrap.style.display = "block";
  statusEl.textContent = "Yükleniyor...";

  task.on("state_changed",
    (snap) => {
      const pct = Math.round((snap.bytesTransferred / snap.totalBytes) * 100);
      progressBar.style.width = pct + "%";
      statusEl.textContent = `Yükleniyor... %${pct}`;
    },
    (err) => { statusEl.textContent = "Hata: " + err.message; },
    async () => {
      const url = await getDownloadURL(task.snapshot.ref);
      await addDoc(collection(db, "videos"), {
        title, category, notes, storagePath, url, createdAt: serverTimestamp()
      });
      statusEl.textContent = "Yüklendi ✓";
      progressWrap.style.display = "none";
      progressBar.style.width = "0%";
      document.getElementById("title").value = "";
      document.getElementById("notes").value = "";
      fileInput.value = "";
    }
  );
};
